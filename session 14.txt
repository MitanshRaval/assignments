#include <stdio.h>

int isEven(int num) {
    if (num % 2 == 0) {
        return 1;
    } else {
        return 0;
    }
}

int main() {
    printf("%d\n", isEven(4));
    printf("%d\n", isEven(7));

    return 0;
}

#include <stdio.h>

void formatFollowersCount(int count) {
    if (count < 1000) {
        printf("%d\n", count);
    } else if (count >= 1000000) {
        printf("%.1fM\n", count / 1000000.0);
    } else {
        printf("%.1fK\n", count / 1000.0);
    }
}

int main() {
    formatFollowersCount(1500);
    formatFollowersCount(1200000);
    formatFollowersCount(850);

    return 0;
}
