#include <stdio.h>
#include <string.h>

struct Playlist
{
    char title[50];
    char artist[50];
    int duration;
};

int main()
{
    struct Playlist song;

    strcpy(song.title, "Tum Hi Ho");
    strcpy(song.artist, "Arijit Singh");
    song.duration = 262;

    printf("Song Title: %s\n", song.title);
    printf("Artist: %s\n", song.artist);
    printf("Duration: %d seconds\n", song.duration);

    return 0;
}

#include <stdio.h>

struct FoodItem
{
    char itemName[50];
    float price;
    float rating;
};

int main()
{
    struct FoodItem menu[3] =
    {
        {"Paneer Pizza", 299.00, 4.5},
        {"Veg Burger", 149.00, 4.2},
        {"Masala Dosa", 120.00, 4.7}
    };

    for (int i = 0; i < 3; i++)
    {
        printf("\nFood Item %d\n", i + 1);
        printf("Name: %s\n", menu[i].itemName);
        printf("Price: ₹%.2f\n", menu[i].price);
        printf("Rating: %.1f\n", menu[i].rating);
    }

    return 0;
}

#include <stdio.h>

struct Time
{
    int hours;
    int minutes;
};

struct MovieShow
{
    char movie[50];
    int screen;
    struct Time time;
};

int main()
{
    struct MovieShow show =
    {
        "Avengers: Endgame",
        3,
        {7, 30}
    };

    printf("Movie: %s, Screen: %d, Time: %02d:%02d\n",show.movie,show.screen,show.time.hours,show.time.minutes);

    return 0;
}

#include <stdio.h>

struct Bio
{
    char description[100];
    int age;
};

struct InstaProfile
{
    char username[50];
    int followers;
    struct Bio bio;
};

int main()
{
    struct InstaProfile profile =
    {
        "mitansh",
        5000,
        {"Just living life my way.", 18}
    };

    printf("Username: %s\n", profile.username);
    printf("Followers: %d\n", profile.followers);
    printf("Bio: %s\n", profile.bio.description);
    printf("Age: %d\n", profile.bio.age);

    return 0;
}
