3(session 7 task 3)

#include <stdio.h>

int main() {

    for (int i = 1; i <= 6; i++) {

        // Print spaces
        for (int j = 1; j <= 6 - i; j++) {
            printf(" ");
        }

        // Print stars
        for (int j = 1; j <= (2 * i - 1); j++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}

int main() {

    for (int i = 0; i < 4; i++) {

        for (int j = 0; j < 4; j++) {

            if ((i + j) % 2 == 0) {
                printf("0 ");

            else {
                printf("1 ");
            }
        }

        printf("\n");

    return 0;
}

#include <stdio.h>

int main() {

    int rows;

    printf("Enter number of rows: ");
    scanf("%d", &rows);

    for (int i = 1; i <= rows; i++) {

        // Print spaces
        for (int j = 1; j <= rows - i; j++) {
            printf(" ");
        }

        // Print stars
        for (int j = 1; j <= (2 * i - 1); j++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}
